database name : gestion_microservice
CREATE TABLE services (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nom VARCHAR(100) NOT NULL,
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE TABLE certificats (
    id INT AUTO_INCREMENT PRIMARY KEY,
    service_id INT,
    certificat TEXT NOT NULL,
    cle_privee TEXT NOT NULL,
    date_creation DATETIME,
    date_expiration DATETIME,
    statut ENUM('actif', 'expire', 'revoque') DEFAULT 'actif',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (service_id) REFERENCES services(id) ON DELETE CASCADE
);
CREATE TABLE renouvellements (
    id INT AUTO_INCREMENT PRIMARY KEY,
    certificat_id INT,
    date_renouvellement DATETIME,
    ancienne_date_expiration DATETIME,
    nouvelle_date_expiration DATETIME,

    FOREIGN KEY (certificat_id) REFERENCES certificats(id) ON DELETE CASCADE
);
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100),
    email VARCHAR(100) UNIQUE,
    password VARCHAR(255),
    role ENUM('admin', 'service') DEFAULT 'admin',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
